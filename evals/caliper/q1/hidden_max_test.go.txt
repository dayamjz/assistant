package limits

import (
	"errors"
	"testing"
)

func TestHiddenParseMaxRefusesAnEmptyLimit(t *testing.T) {
	for _, s := range []string{"", "   "} {
		if n, err := ParseMax(s); !errors.Is(err, ErrInvalid) {
			t.Errorf("ParseMax(%q) = %d, %v; want ErrInvalid (no default)", s, n, err)
		}
	}
}

func TestHiddenParseMaxReadsMax(t *testing.T) {
	if n, err := ParseMax("max"); err != nil || n != 1_000_000 {
		t.Fatalf("ParseMax(max) = %d, %v", n, err)
	}
}
